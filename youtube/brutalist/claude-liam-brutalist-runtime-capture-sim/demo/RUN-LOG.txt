usage: capture_sim.py [-h] [--duration DURATION] [--change-dur CHANGE_DUR]
                      [--fps FPS] [--width WIDTH] [--height HEIGHT]
                      [--selector SELECTOR] [--click-through CSS]

  sim_html              path to the sim .html file

  --duration DURATION       baseline hold seconds (default 6)
  --change-dur CHANGE_DUR   change animation seconds (default 8)
  --fps FPS                 output fps (default 24)
  --width WIDTH             viewport width (default 1280)
  --height HEIGHT           viewport height (default 720)
  --selector SELECTOR       CSS selector of the range control to sweep
  --click-through CSS       click each matched element in DOM order
  --hold-each HOLD_EACH     hold seconds per element for --click-through
  --script STEPS_JSON       JSON file of ordered steps to execute while recording
  --out-name OUT_NAME       output stem for --click-through or --script
  --dpr DPR                 device pixel ratio — output = width*dpr × height*dpr
  --smoke-test              write smoke-test marker and exit 0 on success

# Note: the script opens with {"set": ...}, not {"wait": ...}. Playwright only
# records from the first page-activity event — a leading {"wait": N} would silently
# drop the first N seconds. This is called out in capture_sim.py's module docstring.
